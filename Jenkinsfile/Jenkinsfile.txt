pipeline {
    agent any
    tools {
        maven 'Maven'
        jdk 'JDK25'
    }
    stages {
        stage('环境校验') {
            steps {
                sh 'mvn -v'
                sh 'java -version'
            }
        }
        stage('PMD代码静态检查') {
            steps {
                sh 'mvn pmd:pmd'
            }
            post {
                always { pmd allowNewResults: true, pattern: 'target/pmd.xml' }
            }
        }
        stage('Maven编译打包') {
            steps {
                sh 'mvn clean package -DskipTests'
            }
        }
        stage('单元测试&生成测试报告') {
            steps {
                sh 'mvn test'
            }
            post {
                always {
                    junit 'target/surefire-reports'
                    publishHTML target: [reportDir: 'target/site/jacoco', reportFiles: 'index.html', reportName: '测试覆盖率报告']
                }
            }
        }
        stage('生成JavaDoc文档') {
            steps {
                sh 'mvn javadoc:javadoc jar:jar'
            }
            post {
                always {
                    archiveArtifacts artifacts: 'target/*.jar'
                    publishHTML target: [reportDir: 'target/site/apidocs', reportFiles: 'index.html', reportName: 'JavaDoc文档']
                }
            }
        }
    }
    post {
        success { echo "Jenkins CI流水线全部执行成功" }
        failure { echo "流水线执行失败，请查看日志" }
    }
}
